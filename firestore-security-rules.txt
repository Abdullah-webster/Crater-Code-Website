service cloud.firestore {
  match /databases/{database}/documents {
    // Courses - Public read access, admin write access
    match /courses/{document} {
      allow read: if true;  // Anyone can read courses
      allow write: if request.auth != null 
        && request.auth.token.email == "muhammadali2025222@gmail.com";
    }
    
    // Applications - Public create access, admin read access
    match /applications/{document} {
      allow create: if true;  // Anyone can submit applications
      allow read: if request.auth != null 
        && request.auth.token.email == "muhammadali2025222@gmail.com";
    }
    
    // Deny all other collections
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
